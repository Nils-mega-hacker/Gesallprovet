Closest points calculator(Java)

Implemented a divide-and-conquer algorithm in Java to find the closest pair of points in a 2D point set, using recursive splitting, a brute-force base case, and a y-sorted strip check across the dividing line. Significantly faster than the naive O(n²) approach.


public class ClosestPointsCalculator {
    private static final int BRUTE_FORCE_THRESHOLD = 4;

    public static Point[] findClosestPairOfPoints(Point[] points) {
       Arrays.sort(points, Comparator.comparingInt(Point::x));
       if (points.length <= BRUTE_FORCE_THRESHOLD) {
          return bruteForce(points);
       }
       return divideAndConquer(points);
    }

    private static Point[] divideAndConquer(Point[] points) {
       int mid = points.length / 2;
       Point[] leftBest = findClosestPairOfPoints(Arrays.copyOfRange(points, 0, mid));
       Point[] rightBest = findClosestPairOfPoints(Arrays.copyOfRange(points, mid, points.length));
       double leftDist = leftBest[0].distanceTo(leftBest[1]);
       double rightDist = rightBest[0].distanceTo(rightBest[1]);
       Point[] currentBest = leftDist < rightDist ? leftBest : rightBest;
       double d = Math.min(leftDist, rightDist);
       return stripClosest(points, points[mid].x(), d, currentBest);
    }

    private static Point[] stripClosest(Point[] points, int midX, double d, Point[] currentBest) {
       List strip = new ArrayList<>();
       for (Point p : points) {
          if (Math.abs(p.x() - midX) < d) {
             strip.add(p);
          }
       }
       strip.sort(Comparator.comparingInt(Point::y));
       for (int i = 0; i < strip.size(); i++) {
          for (int j = i + 1; j < strip.size(); j++) {
             if (strip.get(j).y() - strip.get(i).y() >= d) break;
             double dist = strip.get(i).distanceTo(strip.get(j));
             if (dist < d) {
                d = dist;
                currentBest = new Point[]{strip.get(i), strip.get(j)};
             }
          }
       }
       return currentBest;
    }

    private static Point[] bruteForce(Point[] points) {
       Point[] best = {points[0], points[1]};
       for (int i = 0; i < points.length; i++) {
          for (int j = i + 1; j < points.length; j++) {
             if (points[i].distanceTo(points[j]) < best[0].distanceTo(best[1])) {
                best[0] = points[i];
                best[1] = points[j];
             }
          }
       }
       return best;
    }
}
